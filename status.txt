// 這是 MQTT 的基本邏輯架構
void callback(char* topic, byte* payload, unsigned length) {
  char message = (char)payload[0]; // 讀取第一個字元
  if (message == '1') {
    digitalWrite(ledPin, HIGH); // 收到 1 就亮燈
    Serial.println("MQTT 指令：開啟");
  } else {
    digitalWrite(ledPin, LOW);  // 收到 0 就滅燈
    Serial.println("MQTT 指令：關閉");
  }
}
